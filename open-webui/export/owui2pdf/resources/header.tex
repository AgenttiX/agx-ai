\newcommand{\owuiheader}[5]{%
  \stepcounter{owuimsg}%
  \par\Needspace*{5\baselineskip}%
  \pdfbookmark[0]{#5}{owuimsg\theowuimsg}%
  \noindent\begin{tcolorbox}[enhanced,colback=#1,colframe=#1,boxrule=0pt,arc=2pt,
      left=6pt,right=6pt,top=3pt,bottom=3pt,boxsep=0pt,
      before skip=2ex plus .5ex,after skip=1.2ex]
    \textcolor{#2}{\textbf{\sffamily #3}}\hfill{\small\color{owuiMeta}\sffamily #4}%
  \end{tcolorbox}\nobreak\noindent\ignorespaces
}
